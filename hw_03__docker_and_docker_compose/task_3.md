# Подключение к потокам и остановка
docker attach custom-nginx-t2
# Нажать на клавиатуре Ctrl-C.
Увидим логи остановки nginx и возврат в обычную консоль

# Проверка статуса и объяснение
docker ps -a

Контейнер остановился, потому что команда docker attach подключает терминал к основному процессу 
    контейнера (PID 1), которым является nginx. Нажатие Ctrl-C отправляет сигнал SIGINT (прерывание) непосредственно 
    этому процессу. В Docker жизненный цикл контейнера напрямую зависит от его главного процесса: когда PID 1 
    завершает работу, контейнер автоматически останавливается.

# Перезапуск и вход в контейнер
docker start custom-nginx-t2
docker exec -it custom-nginx-t2 bash

# Установка редактора и правка конфига
apt-get update
apt-get install -y nano
nano /etc/nginx/conf.d/default.conf

# Перезагрузка nginx и проверка портов (Внутри контейнера!)
nginx -s reload
curl http://127.0.0.1:80
curl http://127.0.0.1:81

# Выход из контейнера
exit

# Проверка на хост-машине и объяснение проблемы
ss -tlpn | grep 127.0.0.1:8080
docker port custom-nginx-t2
curl http://127.0.0.1:8080

При запуске Docker настроил проброс портов (port binding): он направляет трафик с порта хоста 8080 на порт 
    контейнера 80. Однако мы изменили конфигурацию nginx внутри контейнера, и теперь он слушает порт 81. В 
    результате Docker-прокси успешно доставляет запрос на 80-й порт контейнера, но там больше нет слушающего 
    процесса nginx, что приводит к ошибке соединения (Connection refused).

# Дополнительное задание (Исправление без удаления контейнера)
Чтобы исправить это, не меняя конфиг nginx обратно и не удаляя контейнер, нужно изменить настройки проброса портов 
    на уровне Docker. Это делается через редактирование конфигурационного файла контейнера.
Узнаем ID контейнера:
CONTAINER_ID=$(docker inspect -f '{{.Id}}' custom-nginx-t2)
echo $CONTAINER_ID

Обязательно останавливаем контейнер (Docker не даст изменить файл работающего контейнера):
docker stop custom-nginx-t2

Редактируем файл настроек хоста (потребуются права root):
sudo nano /var/lib/docker/containers/$CONTAINER_ID/hostconfig.json

Внутри файла в секции "PortBindings"
"80/tcp":[{"HostIp":"127.0.0.1","HostPort":"8080"}]
Изменить 80/tcp на 81/tcp, чтобы получилось:
"81/tcp":[{"HostIp":"127.0.0.1","HostPort":"8080"}]

Перезапускаем демон Docker, чтобы он перечитал файлы
sudo systemctl restart docker

Запускаем контейнер и проверяем:
docker start custom-nginx-t2
curl http://127.0.0.1:8080

Так не сработало
Значит попробуем через iptables

# Перенаправляем трафик с 8080 на 80 порт контейнера
# Сначала узнаем IP контейнера
docker inspect -f '{{range .NetworkSettings.Networks}}{{.IPAddress}}{{end}}' custom-nginx-t2

# Затем добавляем правило iptables (замените 172.17.0.X на IP из предыдущей команды)
sudo iptables -t nat -A DOCKER -p tcp --dport 8080 -j DNAT --to-destination 172.17.0.X:81

Тоже не сработало, тогда следующий вариант
# Остановить контейнер
docker stop custom-nginx-t2

# Создать новый образ из текущего контейнера (сохраняет все изменения, включая порт 81 в nginx)
docker commit custom-nginx-t2 custom-nginx-fixed

# Удалить старый контейнер
docker rm custom-nginx-t2

# Запустить НОВЫЙ контейнер с правильным маппингом 8080 (хост) -> 81 (контейнер)
docker run -d --name custom-nginx-t2 -p 127.0.0.1:8080:81 custom-nginx-fixed

# Проверить
docker ps
curl http://127.0.0.1:8080
